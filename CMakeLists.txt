cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" LINFXP_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
project(LinFXP VERSION ${LINFXP_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Concurrent)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SSH REQUIRED IMPORTED_TARGET libssh>=0.11)
pkg_check_modules(SECRET REQUIRED IMPORTED_TARGET libsecret-1)
add_executable(linfxp src/main.cpp src/ssh.cpp src/window.cpp src/transfer.cpp resources.qrc)
target_link_libraries(linfxp PRIVATE Qt6::Widgets Qt6::Concurrent PkgConfig::SSH PkgConfig::SECRET)
target_compile_definitions(linfxp PRIVATE QT_NO_KEYWORDS)
target_compile_definitions(linfxp PRIVATE LINFXP_VERSION="${PROJECT_VERSION}")
target_compile_options(linfxp PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS linfxp RUNTIME DESTINATION bin)
install(FILES packaging/linfxp.desktop DESTINATION share/applications)
install(FILES README.md CHANGELOG.md DESTINATION share/doc/linfxp)
install(FILES screenshots/Screenshot.png DESTINATION share/doc/linfxp/screenshots)
install(FILES LICENSE DESTINATION share/licenses/linfxp)
include(CTest)
if(BUILD_TESTING)
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(test_profiles tests/test_profiles.cpp src/window.cpp src/ssh.cpp src/transfer.cpp resources.qrc)
target_include_directories(test_profiles PRIVATE src)
target_compile_definitions(test_profiles PRIVATE QT_NO_KEYWORDS)
target_compile_options(test_profiles PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(test_profiles PRIVATE Qt6::Widgets Qt6::Concurrent Qt6::Test PkgConfig::SSH PkgConfig::SECRET)
add_test(NAME connection-profiles COMMAND test_profiles)
set_tests_properties(connection-profiles PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=none;QT_STYLE_OVERRIDE=Fusion")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME remote-worker COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
set_tests_properties(remote-worker PROPERTIES
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    ENVIRONMENT "LINFXP_TEST_BUILD_DIR=${CMAKE_BINARY_DIR}")
endif()
